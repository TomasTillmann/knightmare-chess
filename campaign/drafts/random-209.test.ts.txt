// The only variant movement powers here are the cursed Queen and its Pawn composite.
function geometry(model: GameState, piece: PieceState, to: string, capture: boolean): boolean {
  if (!piece.square || to === piece.square) return false
  const [x, y] = xy(piece.square), [tx, ty] = xy(to), dx = tx - x, dy = ty - y
  if (model.effects.some(effect => effectType(effect) === 'forbidden-city') && to === 'b5') return false
  if (model.underElfHill?.some(entry => entry.returned && entry.pieceId === piece.id)) return false
  const composite = model.effects.some(effect => effectType(effect) === 'confabulation') && piece.id === 'white-pawn-c2'
  const cursed = piece.id === 'white-queen-d1' && model.effects.some(effect => effectType(effect) === 'curse') || composite
  if (cursed && Math.max(Math.abs(dx), Math.abs(dy)) > 2) return false
  const roles = composite ? ['pawn', 'queen'] : [piece.role]
  return roles.some(role => {
    if (role === 'knight') return Math.abs(dx) * Math.abs(dy) === 2
    if (role === 'king') return Math.max(Math.abs(dx), Math.abs(dy)) === 1
    if (role === 'pawn') {
      const forward = piece.owner === 'white' ? 1 : -1
      if (capture) return Math.abs(dx) === 1 && dy === forward
      if (dx) return false
      if (dy === forward) return true
      return dy === 2 * forward && (piece.owner === 'white' ? y <= 1 : y >= 6)
        && !model.pieces.some(p => p.square === String.fromCharCode(x + 97) + (y + forward + 1))
    }
    const aligned = role === 'bishop' ? Math.abs(dx) === Math.abs(dy)
      : role === 'rook' ? dx === 0 || dy === 0 : dx === 0 || dy === 0 || Math.abs(dx) === Math.abs(dy)
    if (!aligned) return false
    for (let n = 1; n < Math.max(Math.abs(dx), Math.abs(dy)); n++) {
      const intermediate: string = String.fromCharCode(x + n * Math.sign(dx) + 97) + (y + n * Math.sign(dy) + 1)
      if (model.pieces.some(p => p.square === intermediate) || intermediate === 'b5' && model.effects.some(e => effectType(e) === 'forbidden-city')) return false
    }
    return true
  })
}

function rawCheck(model: GameState, color: Color): boolean {
  const king = model.pieces.find(piece => piece.owner === color && piece.royal && piece.zone === 'board')
  return !!king && model.pieces.some(piece => piece.zone === 'board' && piece.owner !== color && geometry(model, piece, king.square!, true))
}

function board(model: GameState): string {
  const letters = { pawn: 'p', knight: 'n', bishop: 'b', rook: 'r', queen: 'q', king: 'k' }
  return Array.from({ length: 8 }, (_, i) => {
    let rank = '', empty = 0
    for (let file = 0; file < 8; file++) {
      const piece = model.pieces.find(p => p.square === String.fromCharCode(97 + file) + (8 - i))
      if (!piece) { empty++; continue }
      if (empty) rank += empty
      empty = 0
      const letter = letters[piece.role]
      rank += piece.owner === 'white' ? letter.toUpperCase() : letter
    }
    return rank + (empty || '')
  }).join('/')
}
